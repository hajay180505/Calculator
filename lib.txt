#db
from sqlalchemy.orm import sessionmaker
from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base

db_url = "sqlite:///./flight.db"
engine = create_engine(db_url, connect_args={"check_same_thread" : False})
SessionLocal = sessionmaker(bind=engine, autoflush=False, autocommit=False)
Base = declarative_base()

#models
from sqlalchemy import Column, Integer, String
from guleba.flight_service.db import Base

class Flight(Base):
    __tablename__ = "flights"
    id = Column(Integer, primary_key=True, index=True)
    source = Column(String)
    destination = Column(String)

#schemas
  from typing import Optional
from pydantic import BaseModel

class FlightCreate(BaseModel):
    source : str 
    destination : str 
    
    class Config:
        from_attributes = True

class FlightResponse(BaseModel):
    id: int
    source : str 
    destination : str 
    
    class Config:
        from_attributes = True
        
class FlightUpdate(BaseModel):
    source : Optional[str] 
    destination : Optional[str] 
    
    class Config:
        from_attributes = True

#main
from fastapi import Depends, HTTPException, FastAPI
from sqlalchemy.orm import Session
from guleba.flight_service.db import *
from guleba.flight_service.models import *
from guleba.flight_service.schemas import *

Base.metadata.create_all(bind=engine)
app = FastAPI()

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()
        


@app.get("/filghts/{flight_id}", response_model = FlightResponse)
def get_flight(flight_id: int, db: Session = Depends(get_db)):
    res = db.query(Flight).get(flight_id)
    if not res:
        return HTTPException(status_code = 404, detail="Flight not found")
    return res

@app.get("/filghts", response_model = list[FlightResponse])
def get_all_flights(db: Session = Depends(get_db)):
    res = db.query(Flight).all()
    return res

@app.get("/flights/source/{source}", response_model=list[FlightResponse])
def get_flights_by_source(source: str, db: Session = Depends(get_db)):
    res = db.query(Flight).filter(Flight.source == source)
    return res
    
@app.get("/flights/destination/{dest}", response_model=list[FlightResponse])
def get_flights_by_dest(dest: str, db: Session = Depends(get_db)):
    res = db.query(Flight).filter(Flight.destination == dest)
    return res
    
@app.post("/flights", response_model = FlightResponse)
def create_flight(flight: FlightCreate, db: Session = Depends(get_db)):
    res = Flight(**flight.dict())
    db.add(res)
    db.commit()
    db.refresh(res)
    return res
    
@app.delete("/filghts/{flight_id}", response_model = dict)
def get_flight(flight_id: int, db: Session = Depends(get_db)):
    res = db.query(Flight).get(flight_id)
    if not res:
        return HTTPException(status_code = 404, detail="Flight not found")
    db.delete(res)
    db.commit()
    return {'message' : "Flight deleted successfully"}
    
    
@app.put("/filghts/{flight_id}", response_model = FlightResponse)
def get_flight(flight_id: int, flight_data: FlightUpdate, db: Session = Depends(get_db)):
    res = db.query(Flight).get(flight_id)
    if not res:
        return HTTPException(status_code = 404, detail="Flight not found")
    for k,v in flight_data.dict(exclude_unset = True).items():
        setattr(res, k, v)
    db.commit()
    db.refresh(res)
    return res
